     ```dart
     class AiSseClient {
       final String baseUrl;
       final Future<String?> Function() getAccessToken;

       AiSseClient({required this.baseUrl, required this.getAccessToken});

       Stream<SseEvent> streamChatTurn({
         required String path,  // e.g., '/api/v1/ai/intake/message'
         required Map<String, dynamic> body,
       }) async* {
         final token = await getAccessToken();
         // Use flutter_client_sse or dart:io HttpClient
         // Parse SSE events into SseEvent objects
         // Yield: SseEvent(event: 'token', data: '{"delta":"text"}')
         //        SseEvent(event: 'tool_call', data: '{"tool":"create_trade_scope","input":{...},"id":"..."}')
         //        SseEvent(event: 'done', data: '{"stop_reason":"end_turn"}')
         //        SseEvent(event: 'error', data: '{"message":"..."}')
       }

       Future<Map<String, dynamic>> post(String path, Map<String, dynamic> body) async {
         // Standard HTTP POST for non-streaming endpoints (start, complete)
         // Uses dart:io HttpClient with Bearer token
       }

       Future<Map<String, dynamic>> get(String path) async {
         // Standard HTTP GET for conversation fetch
       }
     }
     ```
   - `SseEvent` data class: `{String event, String data}`
   - Register in GetIt: `getIt.registerLazySingleton(() => AiSseClient(baseUrl: apiBaseUrl, getAccessToken: () => secureStorage.read(key: 'access_token')))`
   - IMPORTANT: If `flutter_client_sse` does not support POST, fall back to `dart:io` HttpClient:
     ```dart
     final client = HttpClient();
     final request = await client.postUrl(Uri.parse(url));
     request.headers.set('Authorization', 'Bearer $token');
     request.headers.set('Content-Type', 'application/json');
     request.headers.set('Accept', 'text/event-stream');
     request.write(jsonEncode(body));
     final response = await request.close();
     await for (final chunk in response.transform(utf8.decoder)) {
       // Parse SSE lines
       for (final line in chunk.split('\n')) {
         if (line.startsWith('event: ')) currentEvent = line.substring(7).trim();
         if (line.startsWith('data: ')) yield SseEvent(event: currentEvent, data: line.substring(6));
       }
     }
     ```
