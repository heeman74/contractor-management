    @dataclass(frozen=True)
    class RevenueAnchor:

        """The money fields of one invoice or quote: line-item subtotal plus its discount/tax terms."""
        subtotal: Decimal
        discount_type: str | None
        discount_value: Decimal
        tax_rate: Decimal

    @dataclass(frozen=True)
    class AnchorRevenue:
        invoiced_total: Decimal | None = None
        quoted_total: Decimal | None = None

    @dataclass(frozen=True)
    class ResolvedRevenue:
        total: Decimal | None
        basis: str

    @dataclass(frozen=True)
    class MarginInputs:
        revenue: ResolvedRevenue
        cost: Decimal
        unrated_seconds: int = 0
        has_missing_cost_data: bool = False

    @dataclass(frozen=True)
    class MarginFigures:
        revenue: Decimal | None
        revenue_basis: str
        margin: Decimal | None
        margin_percent: Decimal | None
        incomplete: bool
        incomplete_reasons: tuple[str, ...]

    ```python
    def discount_for(amounts: DocumentAmounts) -> Decimal
    def pre_tax_total(amounts: DocumentAmounts) -> Decimal      # subtotal - discount (D-13 revenue)
    def tax_for(amounts: DocumentAmounts) -> Decimal            # tax on the discounted subtotal
    def document_total(amounts: DocumentAmounts) -> Decimal     # pre_tax_total + tax (display total)
    def revenue_from(documents: Iterable[DocumentAmounts]) -> Decimal   # Σ pre_tax_total, quantized to CENTS
    def resolve_anchor_revenue(anchor: AnchorRevenue) -> ResolvedRevenue
    def combine_revenue_bases(bases: Iterable[str]) -> str
    def missing_cost_data(cost: Decimal, revenue: Decimal | None) -> bool
    def margin_percent_for(margin: Decimal, revenue: Decimal) -> Decimal | None
    def summarize_margin(inputs: MarginInputs) -> MarginFigures
    ```

    Exact semantics:
    - `discount_for`: percent → `(subtotal * discount_value / PERCENT_MULTIPLIER).quantize(CENTS)`
      with NO explicit rounding argument (bit-for-bit identical to the shipped schema code —
      adding ROUND_HALF_UP here would silently change existing invoice/quote totals);
      fixed → `min(discount_value, subtotal)`; anything else → `ZERO_MONEY`.
    - `tax_for`: `(pre_tax_total(amounts) * amounts.tax_rate / PERCENT_MULTIPLIER).quantize(CENTS)`,
      likewise with no explicit rounding argument.
    - `resolve_anchor_revenue`: `invoiced_total is not None` → (invoiced_total, INVOICED);

      Never sums the two, never takes max() (D-01).
    - `combine_revenue_bases`: drop NONE entries; empty → NONE; exactly one distinct → that one;
      more than one → MIXED.
    - `missing_cost_data`: `revenue is not None and revenue > ZERO_MONEY and cost <= ZERO_MONEY`.
    - `margin_percent_for`: `None` when `revenue <= 0`; else
      `(margin / revenue * PERCENT_MULTIPLIER).quantize(PERCENT_PLACES, rounding=ROUND_HALF_UP)`.
    - `summarize_margin`:
        * `inputs.revenue.total is None` → `MarginFigures(None, REVENUE_BASIS_NONE, None, None, False, ())`

          then `INCOMPLETE_NO_COST_DATA` when `has_missing_cost_data`;
          `incomplete = bool(reasons)`.
