```json
"margin": {
  "revenue": "20000.00",

  "incomplete": false,
```

Present on `GET /jobs/{id}/cost-breakdown`, `GET /trade-scopes/{id}/cost-breakdown`, and
`GET /projects/{id}/cost-entries`. `revenue`/`margin`/`margin_percent` are nullable strings;
`revenue_basis` ∈ `"invoiced" | "quoted" | "mixed" | "none"`;
`incomplete_reasons` ⊆ `["unrated_labor", "no_cost_data"]`.

```ts
export interface CostBreakdown {

}
export interface ProjectCostRollup {
  projectId: string; total: string; entries: CostEntry[];
  categories: CategoryTotal[]; labor: LaborCostSummary | null; grandTotal: string | null;
```

    ```ts
      revenue: string | null;
      revenue_basis: string;
      margin: string | null;
      margin_percent: string | null;
      incomplete: boolean;
      incomplete_reasons: string[];

      raw: MarginSummaryApiResponse | null | undefined
    ): MarginSummary | null {
      if (!raw) return null;

        revenue: raw.revenue,

        incomplete: raw.incomplete,

      };
    }
    ```
