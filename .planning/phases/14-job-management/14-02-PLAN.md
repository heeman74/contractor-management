PATCH  /api/v1/jobs/{id}/transition   → Job (body: { new_status, reason?, version })
GET    /api/v1/jobs/{id}/time-entries → TimeEntryResponse[] (if endpoint exists; graceful 404 fallback)
```

Transition rules (backend enforced):
- quote → scheduled, cancelled
- scheduled → in_progress, quote (revert), cancelled
- in_progress → complete, scheduled (revert), cancelled
- complete → invoiced, in_progress (revert)
- invoiced → (terminal — no forward)
- cancelled → (terminal — no forward)
```

    const transitionMutation = useMutation({
      mutationFn: (data: JobTransitionRequest) =>
        apiPatch<Job>(`/api/v1/jobs/${jobId}/transition`, data),
      onSuccess: (updatedJob) => {
        queryClient.setQueryData(["job", jobId], updatedJob);
        toast.success(`Job marked ${updatedJob.status.replace(/_/g, " ")}`);
        setTransitionError(null);
      },
      onError: (err: Error) => {
        const apiErr = err as ApiError;
        if (apiErr.status === 409) {
          queryClient.invalidateQueries({ queryKey: ["job", jobId] });
          setTransitionError("This job was updated by someone else. Refreshing to get the latest version");
        } else {
          setTransitionError(apiErr.detail ?? "Transition failed");
        }
      },
    });
