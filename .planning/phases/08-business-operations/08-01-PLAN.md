    3. **quote_templates** — id (UUID PK), company_id (FK companies), name TEXT NOT NULL, description TEXT NULLABLE, line_items_json TEXT NOT NULL (JSON array of line item definitions), tax_rate NUMERIC(5,2) DEFAULT 0, version/created_at/updated_at/deleted_at.

    4. **invoices** — id (UUID PK), company_id (FK companies), job_id (FK jobs), quote_id (FK quotes NULLABLE), invoice_number TEXT NOT NULL, status TEXT CHECK ('unpaid','partially_paid','paid') DEFAULT 'unpaid', tax_rate NUMERIC(5,2) DEFAULT 0, discount_type TEXT NULLABLE, discount_value NUMERIC(10,2) DEFAULT 0, due_date DATE NULLABLE, issued_at TIMESTAMPTZ NOT NULL, finalized_at TIMESTAMPTZ NULLABLE, version/created_at/updated_at/deleted_at. UNIQUE(company_id, invoice_number).

    5. **invoice_line_items** — id (UUID PK), company_id (FK companies), invoice_id (FK invoices ON DELETE CASCADE), item_type TEXT CHECK ('labor','material'), description TEXT NOT NULL, quantity NUMERIC(10,3) NOT NULL, unit TEXT NOT NULL, unit_price NUMERIC(10,2) NOT NULL, sort_order INT DEFAULT 0, version/created_at/updated_at/deleted_at.

    6. **ALTER TABLE companies** ADD COLUMN invoice_prefix TEXT DEFAULT 'INV', ADD COLUMN invoice_sequence INTEGER DEFAULT 0.

    - QuoteLineItemResponse(BaseResponseSchema)
    - QuoteTemplateCreate/QuoteTemplateResponse
    - InvoiceLineItemCreate, InvoiceCreate, InvoiceUpdate, InvoiceResponse
