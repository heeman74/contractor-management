```dart
void _handleApprove() async {
  setState(() => _isSubmitting = true);
  try {
    final dao = ref.read(taskInspectionDaoProvider);
    await dao.createInspection(TaskInspectionsCompanion.insert(
      id: Value(const Uuid().v4()),
      companyId: authState.companyId,
      taskId: task.id,
      inspectorId: authState.userId,
      decision: 'approved',
      checklistResults: Value(jsonEncode(_checklistResults)),
      createdAt: DateTime.now(),
      updatedAt: DateTime.now(),
    ));
    if (mounted) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Task approved.')));
      Navigator.of(context).pop();
    }
  } catch (e) {
    if (mounted) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Failed to approve task. Please try again.')));
      setState(() => _isSubmitting = false);
    }
  }
}
```

5. **_handleReject:**
```dart
void _handleReject() async {
  final result = await showRejectionSheet(context);
  if (result == null) return;
  setState(() => _isSubmitting = true);
  try {
    final dao = ref.read(taskInspectionDaoProvider);
    await dao.createInspection(TaskInspectionsCompanion.insert(
      id: Value(const Uuid().v4()),
      companyId: authState.companyId,
      taskId: task.id,
      inspectorId: authState.userId,
      decision: 'rejected',
      checklistResults: Value(jsonEncode(_checklistResults)),
      rejectionReason: Value(result['reason'] as String?),
      rejectionComment: Value(result['comment'] as String?),
      createdAt: DateTime.now(),
      updatedAt: DateTime.now(),
    ));
    // Also update task status locally to 'rejected'
    final taskDao = ref.read(taskDaoProvider);
    await taskDao.updateTaskStatus(task.id, 'rejected');
    if (mounted) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Task rejected. Contractor notified.')));
      Navigator.of(context).pop();
    }
  } catch (e) {
    if (mounted) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Failed to submit rejection. Please try again.')));
      setState(() => _isSubmitting = false);
    }
  }
}
```
