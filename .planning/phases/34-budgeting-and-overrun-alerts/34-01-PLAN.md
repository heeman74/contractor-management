```python
class BudgetThreshold(StrEnum):
    warning = "warning"
    overrun = "overrun"

@dataclass(frozen=True)
class BudgetFacts:
    project_name: str
    trade_name: str | None      # None = project budget
    spent: Decimal
    total: Decimal

def percent_used(spent: Decimal, total: Decimal) -> Decimal          # 1dp, ROUND_HALF_UP
def crossed_thresholds(spent: Decimal, total: Decimal) -> tuple[BudgetThreshold, ...]
def format_alert_money(amount: Decimal) -> str                        # "$8,200" / "$8,200.50"
def format_alert_percent(percent: Decimal) -> str                     # "82" / "82.5" / "112"
def budget_alert_text(facts: BudgetFacts, threshold: BudgetThreshold) -> str
def push_title_for(threshold: BudgetThreshold) -> str
```
