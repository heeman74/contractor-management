    ```typescript
    export type ThreadType = 'scope' | 'project_wide';
    export type AttachmentType = 'photo' | 'pdf' | 'annotated_photo';
    export type MessageStatus = 'pending' | 'sent' | 'read' | 'failed';

    export interface ChatThread {
      id: string;
      project_id: string;
      thread_type: ThreadType;
      trade_scope_id: string | null;
      name: string;
      created_at: string;
      last_message?: ChatMessage | null;
      unread_count: number;
      member_count: number;
    }

    export interface ChatMessage {
      id: string;
      thread_id: string;
      sender_id: string;
      sender_name: string;
      content: string | null;
      seq: number;
      attachment_url: string | null;
      attachment_type: AttachmentType | null;
      annotation_data: string | null;
      mentions: string[];  // user IDs
      mention_all: boolean;
      created_at: string;
    }

    export interface ChatReadReceipt {
      user_id: string;
      user_name: string;
      last_read_seq: number;
      read_at: string;
    }

    export interface SendMessagePayload {
      id: string;
      content?: string;
      attachment_url?: string;
      attachment_type?: AttachmentType;
      annotation_data?: string;
      mentions?: string[];
      mention_all?: boolean;
    }

    // WebSocket incoming event types
    export type WsEvent =
      | { type: 'message'; message: ChatMessage }
      | { type: 'typing'; user_id: string; user_name: string }
      | { type: 'read_receipt'; user_id: string; last_read_seq: number }
      | { type: 'pong' };
    ```

    2. **useChatWebSocket.ts**:
    - `useChatWebSocket(threadId: string | null)` — returns `{ sendMessage, sendTyping, sendRead, connectionState }`
    - Uses `useRef<WebSocket>` for connection (not useState — avoids re-renders per RESEARCH.md pattern)
    - `useEffect` on threadId: create WebSocket to `ws://{apiHost}/api/v1/ws/chat/{threadId}?token={accessToken}`
    - Get token from cookie or auth context (use the same approach as existing apiClient for auth)
    - Reconnect with exponential backoff: 1s -> 2s -> 4s -> 8s -> 16s -> 32s -> 64s cap
    - `connectionState`: 'connected' | 'reconnecting' | 'disconnected'
    - On message: parse JSON, call `onMessage` callback (passed as option or dispatched via custom event/callback ref)
    - On reconnect: invalidate TanStack Query `['chat-messages', threadId]` to fetch missed messages
    - Heartbeat: send `{"type": "ping"}` every 30s; reconnect if no pong in 10s
    - `sendMessage(payload: SendMessagePayload)` — send `{"type": "message", ...payload}` over WS
    - `sendTyping()` — send `{"type": "typing"}` (debounced externally)
    - `sendRead(seq: number)` — send `{"type": "read", "seq": seq}`
    - Cleanup: close WS on unmount

    3. **useChatMessages.ts**:
    - `useChatMessages(threadId: string | null)` — returns `{ messages, fetchOlderMessages, hasMore, isLoading }`
    - Uses `useInfiniteQuery` from TanStack Query with cursor-based pagination:
      - queryKey: `['chat-messages', threadId]`
      - queryFn: `apiClient(/api/v1/chat/threads/${threadId}/messages?before_seq=${pageParam}&limit=50)`
      - getNextPageParam: if returned messages.length === 50, next cursor is `messages[0].seq` (lowest seq)
      - Flatten pages into single sorted array (by seq ASC)
    - `appendMessage(msg: ChatMessage)` — optimistically add to query cache via `queryClient.setQueryData`
    - `fetchOlderMessages()` — call `fetchNextPage()`
    - `hasMore` — `hasNextPage` from useInfiniteQuery
    - WebSocket new messages integrated by calling `appendMessage` when useChatWebSocket fires a 'message' event
