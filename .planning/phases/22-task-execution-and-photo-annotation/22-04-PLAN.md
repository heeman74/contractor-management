   export type AnnotationTool = 'arrow' | 'circle' | 'text' | 'measurement';

   export interface Annotation {
     id: string;
     tool: AnnotationTool;
     color: string;        // hex e.g. "#FF0000"
     thickness: number;    // stroke width pixels
     startX?: number;      // normalized 0-1
     startY?: number;
     endX?: number;
     endY?: number;
     x?: number;           // top-left for circle
     y?: number;
     width?: number;
     height?: number;
     label?: string;       // text content or measurement value

   }

     version: 1;
     canvasWidth: number;
     canvasHeight: number;
     annotations: Annotation[];
   }

2. Create `web/src/features/tasks/hooks/usePhotoAnnotation.ts`:
   - `usePhotoAnnotation(initialAnnotations?: AnnotationLayer)` hook
   - State: `annotationsRef = useRef<Annotation[]>(initialAnnotations?.annotations ?? [])`, `canvasRef = useRef<HTMLCanvasElement>(null)`, `activeToolRef = useRef<AnnotationTool>('arrow')`, `activeColorRef = useRef('#FF0000')`, `redrawTrigger` (useState counter to force re-renders)
   - `addAnnotation(ann: Annotation)`: push to annotationsRef.current, increment redrawTrigger
   - `undoAnnotation()`: pop last from annotationsRef.current, increment redrawTrigger
   - `clearAnnotations()`: clear array, increment
   - `setActiveTool(tool: AnnotationTool)`: update activeToolRef
   - `setActiveColor(color: string)`: update activeColorRef
   - `getAnnotationLayer(canvasWidth: number, canvasHeight: number) -> AnnotationLayer`: return { version: 1, canvasWidth, canvasHeight, annotations: annotationsRef.current }
   - `useEffect` on [redrawTrigger]: call `redrawCanvas(canvasRef.current, annotationsRef.current)` which clears + redraws all annotations

   - `redrawCanvas(canvas, annotations)` function:
     - ctx = canvas.getContext('2d')
     - ctx.clearRect(0, 0, canvas.width, canvas.height)
     - For each annotation: call drawAnnotation(ctx, ann, canvas.width, canvas.height)

   - `drawAnnotation(ctx, ann, w, h)`:
     - Denormalize: startX * w, startY * h, etc.
     - ctx.strokeStyle = ann.color, ctx.lineWidth = ann.thickness
     - Arrow: drawLine + arrowhead triangle (rotate at angle of line)
     - Circle: ctx.beginPath, ctx.ellipse(centerX, centerY, radiusX, radiusY, 0, 0, 2*PI), ctx.stroke
     - Text: ctx.font = `${ann.fontSize || 16}px sans-serif`, ctx.fillStyle = ann.color, ctx.fillText(ann.label, x*w, y*h)
     - Measurement: drawLine + tick marks (short perpendicular lines at endpoints) + ctx.fillText for label at midpoint

     - `<img>` element (loads imageUrl, onLoad sets canvas dimensions to match image natural dimensions)

   - Mode state: 'pan' | 'draw' (useState)

   - Active tool: `aria-pressed="true"` on selected tool button
   - Save: calls `onSave(JSON.stringify(getAnnotationLayer(canvas.width, canvas.height)))`
   - Discard: calls `onDiscard()`
   - All buttons use shadcn Button component with variant="outline" for tools, variant="default" for save, variant="ghost" for discard
