class FiredFinding:
    """One finding that just alerted — the hand-off to FCM dispatch, resolved in the
    request session so the background task receives primitives only."""
    alert_id: uuid.UUID
    finding_id: uuid.UUID
    company_id: uuid.UUID
    project_id: uuid.UUID

async def _fire_finding(self, published: PublishedFinding) -> FiredFinding | None:
    """Claim this finding's alert and write it; None when it already alerted."""
    if await self.repository.claim_alert(published.finding_id) is None:
        return None
    alert = await AlertRepository(self.db).create(self._build_alert(published))
    ... set finding.dashboard_alert_id = alert.id ...
    return FiredFinding(..., title=PUSH_TITLE_BY_BAND[published.severity_band],

Then wire the lifecycle into the company entry point:
```python
async def analyze_company(self, *, company_id: uuid.UUID, target_date: date) -> None:
    """The scheduler contract: method(company_id=..., target_date=...).

    Resolve-then-claim ordering is load-bearing: the band lives inside the
    fingerprint, so a worsening band is a NEW fingerprint — and the prior band's row
    must resolve in the same run or both stay open (D-06).
    """
    result = await self.publish_findings(company_id, target_date)
    # KEEP-SET = every fingerprint that still QUALIFIES tonight, NOT what published.
    await self.repository.resolve_absent_fingerprints(result.qualifying_fingerprints)
    fired = [f for f in [await self._fire_finding(p) for p in result.published] if f is not None]
    recipients = await self._recipients_for(company_id)
    self._schedule_profitability_pushes(fired, recipients)
```

**The keep-set is `result.qualifying_fingerprints`, never `[p.fingerprint for p in
result.published]`.** `publish_findings` drops a candidate from `published` on three
paths — Claude raised, the draft failed the length contract, or the nightly cap hit —
and none of those mean the erosion cleared. Resolving such a row would let the next
successful night insert a fresh row with `alerted_at IS NULL`, claim it, and fire a
SECOND alert for a condition that never cleared: a direct D-06 violation ("a new alert
fires only when the condition clears and recurs, or worsens into a different band") and
exactly the over-firing RESEARCH Pitfall 6 warns gets all alerts ignored.
