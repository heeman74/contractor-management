```python
@dataclass(frozen=True)
class QuoteGapInputs:
    """The three per-anchor maps the D-03 signal-3 comparison needs, all supplied
    by the caller from already-fetched rows."""
    resolved: Mapping[RevenueAnchor, ResolvedRevenue]
    latest_quotes: Mapping[RevenueAnchor, DocumentAmounts]
    anchor_costs: Mapping[RevenueAnchor, Decimal]

@dataclass(frozen=True)
class QuoteGap:
    """The signal-3 result, every field a figure the AI is allowed to cite."""
    points: Decimal
    billed_margin_percent: Decimal
    quote_implied_margin_percent: Decimal
    over_quote_dollars: Decimal

def latest_quote_per_anchor(
    rows: Sequence[tuple[RevenueAnchor, DocumentAmounts]],
) -> dict[RevenueAnchor, DocumentAmounts]:
    """The newest approved quote per anchor, INCLUDING anchors that have invoices.

    anchor_revenues() cannot supply this leg: it discards quotes at invoiced anchors
    by design (D-01 invoices-win-outright) — which is exactly the leg this signal
    compares against. Rows arrive newest-first per anchor, so setdefault wins.
    """

def quote_implied_gap(inputs: QuoteGapInputs) -> QuoteGap | None:
    """Percentage points by which billed margin sits below quote-implied margin,
    across the anchors both sides share. None when the comparison is not meaningful.
    """
```
`quote_implied_gap` steps, each a named local or a small private helper so the
function stays under 20 lines:
1. `comparable = resolved.keys() & latest_quotes.keys()`; return None if empty.
2. Tautology guard: return None unless at least one comparable anchor's
   `resolved[anchor].basis == REVENUE_BASIS_INVOICED` (Pitfall 5 — a quote-only
   project's gap is identically zero and a rounding artifact could fire it).
3. `cost = sum(inputs.anchor_costs.get(anchor, ZERO_MONEY) for anchor in comparable)`
   — the caller builds `anchor_costs` with the promoted `contributing_anchor_cost`,
   so job-anchored derived labor is included exactly once.
4. `billed = sum(resolved[a].total for a in comparable if ... is not None)`;
   `implied = sum(quoted_revenue(latest_quotes[a]) for a in comparable)`.
5. Both percents via `margin_percent_for(revenue - cost, revenue)`; return None if
   either is None.
6. Return `QuoteGap(points=implied_pct - billed_pct, ..., over_quote_dollars=(implied - billed).quantize(CENTS))`.
