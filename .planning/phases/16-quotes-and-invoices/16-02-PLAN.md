const { data: quote, isLoading, error } = useQuery({
  queryKey: ["quote", id],

// Fetch job for client info, job description, and job status (needed for Generate Invoice button)
const { data: job } = useQuery({
  queryKey: ["job", quote?.job_id],

// Fetch linked invoice if quote is approved (for sidebar linked invoice card)
const { data: linkedInvoice } = useQuery({

});
