   ```python
   """Phase 17 CRM E2E integration tests — tests CRM router against real database."""
   import pytest
   import uuid
   from httpx import AsyncClient

   pytestmark = pytest.mark.asyncio

   class TestCRMClientList:
       """CRM-01: Admin can view a searchable list of all clients."""

       async def test_list_clients_returns_200(self, client: AsyncClient, admin_token: str):
           """GET /api/v1/crm/clients returns 200 with list."""
           response = await client.get(
               "/api/v1/crm/clients",
               headers={"Authorization": f"Bearer {admin_token}"},
           )
           assert response.status_code == 200
           data = response.json()
           assert isinstance(data, list)

       async def test_list_clients_with_search(self, client: AsyncClient, admin_token: str):
           """GET /api/v1/crm/clients?search=xyz filters results."""
           response = await client.get(
               "/api/v1/crm/clients?search=nonexistent_search_term_xyz",
               headers={"Authorization": f"Bearer {admin_token}"},
           )
           assert response.status_code == 200
           data = response.json()
           assert isinstance(data, list)
           # With a nonsense search term, should return empty or fewer results
           assert len(data) == 0 or isinstance(data[0], dict)

       async def test_list_clients_pagination(self, client: AsyncClient, admin_token: str):
           """GET /api/v1/crm/clients?offset=0&limit=10 respects pagination."""
           response = await client.get(
               "/api/v1/crm/clients?offset=0&limit=10",
               headers={"Authorization": f"Bearer {admin_token}"},
           )
           assert response.status_code == 200
           data = response.json()
           assert isinstance(data, list)
           assert len(data) <= 10

       async def test_list_clients_requires_auth(self, client: AsyncClient):
           """GET /api/v1/crm/clients without token returns 401."""
           response = await client.get("/api/v1/crm/clients")
           assert response.status_code == 401

   class TestCRMClientDetail:
       """CRM-02: Admin can view client detail with job history."""

       async def test_client_detail_not_found(self, client: AsyncClient, admin_token: str):
           """GET /api/v1/crm/clients/{unknown_id} returns 404."""
           fake_id = str(uuid.uuid4())
           response = await client.get(
               f"/api/v1/crm/clients/{fake_id}",
               headers={"Authorization": f"Bearer {admin_token}"},
           )
           assert response.status_code in (404, 422)  # 404 or validation error

       async def test_client_detail_requires_auth(self, client: AsyncClient):
           """GET /api/v1/crm/clients/{id} without token returns 401."""
           fake_id = str(uuid.uuid4())
           response = await client.get(f"/api/v1/crm/clients/{fake_id}")
           assert response.status_code == 401

       async def test_client_detail_response_shape(self, client: AsyncClient, admin_token: str):
           """If clients exist, detail endpoint returns expected fields."""
           # First get a client from the list
           list_response = await client.get(
               "/api/v1/crm/clients?limit=1",
               headers={"Authorization": f"Bearer {admin_token}"},
           )
           if list_response.status_code == 200 and len(list_response.json()) > 0:
               client_data = list_response.json()[0]
               user_id = client_data["user_id"]
               detail_response = await client.get(
                   f"/api/v1/crm/clients/{user_id}",
                   headers={"Authorization": f"Bearer {admin_token}"},
               )
               assert detail_response.status_code == 200
               detail = detail_response.json()
               # Verify expected fields exist
               assert "user_id" in detail
               assert "email" in detail
               assert "jobs" in detail
               assert isinstance(detail["jobs"], list)
   ```
